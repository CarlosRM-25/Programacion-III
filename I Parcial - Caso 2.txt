using System;
using System.Collections.Generic;

namespace AgendaPro
{
    class Persona
    {
        public int Id { get; set; }
        public string Nombre { get; set; }
        public string Telefono { get; set; }

        public Persona(int id, string nombre, string telefono)
        {
            Id = id;
            Nombre = nombre;
            Telefono = telefono;
        }
    }

    class Cita
    {
        public int PersonaId { get; set; }
        public string Fecha { get; set; }
        public string Descripcion { get; set; }

        public Cita(int personaId, string fecha, string descripcion)
        {
            PersonaId = personaId;
            Fecha = fecha;
            Descripcion = descripcion;
        }
    }

    class Program
    {
        static void Main()
        {
            List<Persona> personas = new List<Persona>();
            List<Cita> citas = new List<Cita>();
            string opcion;

            do
            {
                Console.WriteLine("\n=== SISTEMA DE AGENDA - PyME 'AgendaPro' ===");
                Console.WriteLine("1. Registrar persona");
                Console.WriteLine("2. Listar personas");
                Console.WriteLine("3. Crear cita para una persona");
                Console.WriteLine("4. Listar citas por PersonaId");
                Console.WriteLine("5. Mostrar todas las citas");
                Console.WriteLine("6. Salir");
                Console.Write("Seleccione una opción: ");
                opcion = Console.ReadLine();

                switch (opcion)
                {
                    case "1":
                        RegistrarPersona(personas);
                        break;
                    case "2":
                        ListarPersonas(personas);
                        break;
                    case "3":
                        CrearCita(personas, citas);
                        break;
                    case "4":
                        ListarCitasPorPersona(citas);
                        break;
                    case "5":
                        ListarTodasLasCitas(citas);
                        break;
                    case "6":
                        Console.WriteLine("Saliendo del sistema... ¡Hasta luego!");
                        break;
                    default:
                        Console.WriteLine("⚠️ Opción no válida. Intente de nuevo.");
                        break;
                }

            } while (opcion != "6");
        }

        static void RegistrarPersona(List<Persona> personas)
        {
            try
            {
                Console.Write("Ingrese Id de la persona: ");
                int id = int.Parse(Console.ReadLine());

                // Validar Id único
                if (personas.Exists(p => p.Id == id))
                {
                    Console.WriteLine("⚠️ Ya existe una persona con ese Id.");
                    return;
                }

                Console.Write("Nombre: ");
                string nombre = Console.ReadLine();

                Console.Write("Teléfono: ");
                string telefono = Console.ReadLine();

                personas.Add(new Persona(id, nombre, telefono));
                Console.WriteLine("✅ Persona registrada correctamente.");
            }
            catch
            {
                Console.WriteLine("⚠️ Entrada inválida. El Id debe ser numérico.");
            }
        }

        static void ListarPersonas(List<Persona> personas)
        {
            Console.WriteLine("\n--- LISTA DE PERSONAS ---");
            if (personas.Count == 0)
            {
                Console.WriteLine("No hay personas registradas.");
                return;
            }

            foreach (var p in personas)
            {
                Console.WriteLine($"{p.Id}|{p.Nombre}|{p.Telefono}");
            }
        }

        static void CrearCita(List<Persona> personas, List<Cita> citas)
        {
            try
            {
                Console.Write("Ingrese el Id de la persona para la cita: ");
                int personaId = int.Parse(Console.ReadLine());

                // Validar existencia de la persona
                if (!personas.Exists(p => p.Id == personaId))
                {
                    Console.WriteLine("⚠️ No existe una persona con ese Id.");
                    return;
                }

                Console.Write("Fecha de la cita (dd/mm/aaaa): ");
                string fecha = Console.ReadLine();

                Console.Write("Descripción: ");
                string descripcion = Console.ReadLine();

                citas.Add(new Cita(personaId, fecha, descripcion));
                Console.WriteLine("✅ Cita creada exitosamente.");
            }
            catch
            {
                Console.WriteLine("⚠️ Entrada inválida. Revise los datos ingresados.");
            }
        }

        static void ListarCitasPorPersona(List<Cita> citas)
        {
            try
            {
                Console.Write("Ingrese el Id de la persona: ");
                int personaId = int.Parse(Console.ReadLine());

                Console.WriteLine($"\n--- CITAS DE LA PERSONA {personaId} ---");

                bool hayCitas = false;
                foreach (var c in citas)
                {
                    if (c.PersonaId == personaId)
                    {
                        Console.WriteLine($"{c.PersonaId}|{c.Fecha}|{c.Descripcion}");
                        hayCitas = true;
                    }
                }

                if (!hayCitas)
                    Console.WriteLine("No hay citas registradas para esta persona.");
            }
            catch
            {
                Console.WriteLine("⚠️ Entrada inválida. El Id debe ser numérico.");
            }
        }

        static void ListarTodasLasCitas(List<Cita> citas)
        {
            Console.WriteLine("\n--- TODAS LAS CITAS ---");
            if (citas.Count == 0)
            {
                Console.WriteLine("No hay citas registradas.");
                return;
            }

            foreach (var c in citas)
            {
                Console.WriteLine($"{c.PersonaId}|{c.Fecha}|{c.Descripcion}");
            }
        }
    }
}
